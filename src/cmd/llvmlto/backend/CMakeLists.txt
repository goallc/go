cmake_minimum_required(VERSION 3.20)
project(GoALLCLTO LANGUAGES C CXX)
find_package(LLVM REQUIRED CONFIG)
if(NOT LLVM_VERSION_MAJOR EQUAL 23)
  message(FATAL_ERROR "GoALLCLTO requires LLVM 23")
endif()
list(APPEND CMAKE_MODULE_PATH "${LLVM_CMAKE_DIR}")
include(AddLLVM)
add_executable(goallc-lto main.cpp)
llvm_update_compile_flags(goallc-lto)
target_include_directories(goallc-lto SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
add_definitions(${LLVM_DEFINITIONS})
target_compile_features(goallc-lto PRIVATE cxx_std_17)
target_link_libraries(goallc-lto PRIVATE LLVM)

set(GOALLC_LTO_PLUGIN "" CACHE FILEPATH "GoALLCStatepoints plugin used by tests")
include(CTest)
if(BUILD_TESTING)
  if(NOT EXISTS "${GOALLC_LTO_PLUGIN}")
    message(FATAL_ERROR "BUILD_TESTING requires GOALLC_LTO_PLUGIN")
  endif()
  find_program(FILECHECK FileCheck HINTS ${LLVM_TOOLS_BINARY_DIR} REQUIRED)
  find_program(NOT_TOOL not HINTS ${LLVM_TOOLS_BINARY_DIR} REQUIRED)
  find_program(BASH bash REQUIRED)
  add_test(NAME full-lto-cross-package
    COMMAND ${BASH} -o pipefail -c
      "'$<TARGET_FILE:goallc-lto>' --plugin='${GOALLC_LTO_PLUGIN}' --output-dir='${CMAKE_CURRENT_BINARY_DIR}/test-output' '${CMAKE_CURRENT_SOURCE_DIR}/test/caller.ll' '${CMAKE_CURRENT_SOURCE_DIR}/test/callee.ll' && '${FILECHECK}' '${CMAKE_CURRENT_SOURCE_DIR}/test/caller.ll' --input-file='${CMAKE_CURRENT_BINARY_DIR}/test-output/0.ll' && '${FILECHECK}' '${CMAKE_CURRENT_SOURCE_DIR}/test/callee.ll' --input-file='${CMAKE_CURRENT_BINARY_DIR}/test-output/1.ll'")

  function(add_lto_test NAME)
    set(OUT "${CMAKE_CURRENT_BINARY_DIR}/${NAME}")
    set(COMMAND "'$<TARGET_FILE:goallc-lto>' --plugin='${GOALLC_LTO_PLUGIN}' --output-dir='${OUT}'")
    set(CHECKS "")
    set(INDEX 0)
    foreach(INPUT IN LISTS ARGN)
      set(SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/test/${INPUT}.ll")
      string(APPEND COMMAND " '${SOURCE}'")
      string(APPEND CHECKS " && '${FILECHECK}' '${SOURCE}' --input-file='${OUT}/${INDEX}.ll'")
      math(EXPR INDEX "${INDEX} + 1")
    endforeach()
    add_test(NAME ${NAME} COMMAND ${BASH} -o pipefail -c "${COMMAND}${CHECKS}")
  endfunction()

  function(add_lto_error_test NAME CHECK PREFIX PIPELINE)
    set(COMMAND "'${NOT_TOOL}' '$<TARGET_FILE:goallc-lto>' --plugin='${GOALLC_LTO_PLUGIN}' --output-dir='${CMAKE_CURRENT_BINARY_DIR}/${NAME}' --passes='${PIPELINE}'")
    foreach(INPUT IN LISTS ARGN)
      string(APPEND COMMAND " '${CMAKE_CURRENT_SOURCE_DIR}/test/${INPUT}.ll'")
    endforeach()
    string(APPEND COMMAND " 2>&1 | '${FILECHECK}' '${CMAKE_CURRENT_SOURCE_DIR}/test/${CHECK}.ll' --check-prefix='${PREFIX}'")
    add_test(NAME ${NAME} COMMAND ${BASH} -o pipefail -c "${COMMAND}")
  endfunction()

  add_lto_test(full-lto-package-metadata metadata-caller metadata-callee metadata-leaf)
  add_lto_test(full-lto-bss bss-declaration bss-definition)
  add_lto_error_test(full-lto-bss-too-small bss-too-small ERROR "lto<O2>" bss-declaration bss-too-small)
  add_lto_error_test(full-lto-duplicate-package caller DUPLICATE "lto<O2>" caller caller)
  add_lto_error_test(full-lto-incompatible-config incompatible-config ERROR "lto<O2>" caller incompatible-config)
  add_lto_error_test(full-lto-duplicate-index duplicate-index ERROR "lto<O2>" duplicate-index)
  add_lto_error_test(full-lto-new-definitions global-dtors ERROR lower-global-dtors global-dtors)
  add_lto_error_test(full-lto-input-alias alias ERROR "lto<O2>" alias)
endif()
